% Título do slide: uma linha pequena de localização (\framesubtitle) acima da
% afirmação que o slide sustenta (\frametitle).
\makeatletter
\setbeamertemplate{frametitle}{%
  \nointerlineskip\vskip4.5mm%
  \begin{beamercolorbox}[wd=\paperwidth, leftskip=9mm]{frametitle}%
    \begin{minipage}[t]{\dimexpr\paperwidth-18mm\relax}\raggedright
    \ifx\insertframesubtitle\@empty\else%
      {\usebeamerfont{framesubtitle}\usebeamercolor[fg]{framesubtitle}%
       \insertframesubtitle\par}\vskip1mm%
    \fi%
    {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}%
     \insertframetitle\par}%
    \end{minipage}%
  \end{beamercolorbox}%
  \vskip-3.5mm%
}
\makeatother

% \numero{valor}{o que ele mede}: número em destaque com a explicação abaixo.
% \numerotexto: o mesmo, quando o destaque é uma expressão e não um número.
\newcommand{\numerobase}[3]{%
  \par\noindent
  {\color{destaque}\vrule width 1mm}\hspace{2.5mm}%
  \begin{minipage}[c]{\dimexpr\linewidth-3.5mm\relax}
    \raggedright
    {#1\bfseries\boldmath #2\par}\vspace{1mm}
    {\fontsize{8.5}{10.5}\selectfont\color{tintaSec}#3\par}
  \end{minipage}\par}
\newcommand{\numero}[2]{\numerobase{\fontsize{19}{21}\selectfont}{#1}{#2}}
\newcommand{\numerotexto}[2]{\numerobase{\fontsize{13}{15}\selectfont}{#1}{#2}}

% \nota{texto}: condições de leitura do gráfico, em letra pequena.

\newcommand{\mensagem}[1]{{\par\noindent{\color{grade}\rule{\linewidth}{0.5pt}}\par\vspace{0.8mm}%
