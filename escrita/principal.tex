\renewcommand{\ABNTEXchapterfont}{\rmfamily\bfseries}

\setsecnumformat{\csname the#1\endcsname\space}

% Espaço entre título e texto e entre texto e título seguinte: uma linha em branco
% (itens 2.2 e 2.4). A seção primária começa no alto da página.

\setlength{\afterchapskip}{\onelineskip}

% Sumário: mesmo destaque dos títulos no texto e mesmo tamanho (12). A classe já põe a
% seção primária em negrito e caixa alta; as terciárias e quaternárias saíam em tamanho
% normal e reduzido, sem o destaque do texto.
% A classe aplicava a fonte dos títulos (agora negrito) no início do sumário, fora do
% grupo do título, e o negrito passava a todas as entradas; volta o início padrão do
% memoir, e o título SUMÁRIO continua formatado como as demais seções sem número.

% Listas de ilustrações e de tabelas sem linha em branco entre os grupos de cada seção,
% como nos modelos do manual (itens 4.10 e 4.11). A classe já desativava esse espaço, mas

% --- Equações (item 2.5.3): numeração sequencial em algarismos arábicos, (1), (2)...
% Antes: numeração por seção primária, (4.1), (5.2)...
\counterwithout{equation}{chapter}

% --- Alíneas (item 2.4.1): indicadas por letras minúsculas seguidas de parêntese, a), b),
% com a letra no recuo do parágrafo (2 cm) e a continuação alinhada sob a primeira letra
% do texto da alínea, em espaço 1,5 e sem espaço extra entre elas. Os itens marcados com
% "•" (itemize) passam a ser alíneas; listas dentro de alíneas viram subalíneas, com
% travessão (item 2.4.2). As listas numeradas (enumerate) mantêm os números, porque o
% texto se refere a elas pelo número (por exemplo, "passo 1"), e recebem o mesmo recuo.
\setlist[alineas,1]{label={\alph*)},align=left,labelindent=\parindent,
  labelwidth=\widthof{m)},labelsep=0.5em,leftmargin=!,
  topsep=0pt,partopsep=0pt,itemsep=0pt,parsep=0pt}
\setlist[alineas,2]{label={--},align=left,labelindent=0pt,
  labelwidth=\widthof{--},labelsep=0.5em,leftmargin=!,
  topsep=0pt,partopsep=0pt,itemsep=0pt,parsep=0pt}
\renewenvironment{itemize}{\begin{alineas}}{\end{alineas}}
\setlist[enumerate,1]{align=left,labelindent=\parindent,
  labelwidth=\widthof{(B)},labelsep=0.5em,leftmargin=!,
  topsep=0pt,partopsep=0pt,itemsep=0pt,parsep=0pt}

  pdftitle={\imprimirtitulo},
  pdfauthor={\imprimirautor},
  pdfsubject={\imprimirtipotrabalho}}}

% --- Linhas que ultrapassavam a margem: folga extra de espaçamento entre palavras, só
% usada quando o parágrafo não cabe de outro modo (não altera o texto).
\setlength{\emergencystretch}{2em}

% --- Lista de siglas (item 4.13): título "LISTA DE SIGLAS", porque a lista só tem siglas,
% e itens no formato "SIGLA – expressão por extenso", um por linha, em espaço 1,5.
\renewcommand{\listadesiglasname}{Lista de siglas}
\renewenvironment{siglas}{%
  \pretextualchapter{\listadesiglasname}%
  \begin{list}{}{%
    \setlength{\leftmargin}{0pt}\setlength{\labelwidth}{0pt}%
    \setlength{\labelsep}{0pt}\setlength{\itemindent}{0pt}%
    \setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}%
    \renewcommand{\makelabel}[1]{##1~\textendash~}}%
}{%
  \end{list}%
  \cleardoublepage%
}

% --- Palavras-chave (item 4.9): expressão "Palavras-chave:" sem destaque, como no modelo
% do manual, e linhas seguintes alinhadas com o início da primeira palavra-chave.
% Resumo e abstract com o recuo de parágrafo do texto, como no modelo do manual
% (Figura 22); a classe zerava esse recuo.
\AtBeginDocument{\setlength{\absparindent}{\parindent}}
\newlength{\ufvjmrotulopalavras}
% O resumo do memoir é uma lista, e dentro de listas o \hangindent é ignorado; por isso
% o recuo da segunda linha em diante é feito com \parshape.
\makeatletter
\newcommand{\palavraschave}[2]{%
  \par\noindent
  \settowidth{\ufvjmrotulopalavras}{#1:\space}%
  \parshape 2 \@totalleftmargin \linewidth
    \dimexpr\@totalleftmargin+\ufvjmrotulopalavras\relax
    \dimexpr\linewidth-\ufvjmrotulopalavras\relax
  #1:\space#2\par}
\makeatother
